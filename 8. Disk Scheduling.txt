
#include <stdio.h>
#include <stdlib.h>

#define N 9
#define DISK_SIZE 200

int main()
{
    int req[N] = {55, 58, 39, 18, 90, 160, 150, 38, 184};
    int i, j, head = 100;
    int total, current, distance;
    int visited[N] = {0};
    int sorted[N], movement;

    // SSTF Algorithm
    printf("\n--- SSTF Algorithm ---\n");

    total = 0;
    current = head;

    printf("Seek Sequence: %d", current);

    for(i = 0; i < N; i++)
    {
        int min = 9999;
        int index = -1;

        for(j = 0; j < N; j++)
        {
            if(!visited[j])
            {
                distance = abs(current - req[j]);

                if(distance < min)
                {
                    min = distance;
                    index = j;
                }
            }
        }

        visited[index] = 1;
        total += min;
        current = req[index];

        printf(" -> %d", current);
    }

    printf("\nTotal Head Movement = %d", total);
    printf("\nAverage Seek Length = %.2f\n",
           (float)total / N);

    // Sort requests in ascending order
    for(i = 0; i < N; i++)
        sorted[i] = req[i];

    for(i = 0; i < N - 1; i++)
    {
        for(j = 0; j < N - i - 1; j++)
        {
            if(sorted[j] > sorted[j + 1])
            {
                int temp = sorted[j];
                sorted[j] = sorted[j + 1];
                sorted[j + 1] = temp;
            }
        }
    }

    // SCAN Algorithm (Direction towards 0)
    printf("\n--- SCAN Algorithm ---\n");

    total = 0;
    current = head;

    printf("Seek Sequence: %d", current);

    for(i = N - 1; i >= 0; i--)
    {
        if(sorted[i] < head)
        {
            movement = abs(current - sorted[i]);
            total += movement;
            current = sorted[i];

            printf(" -> %d", current);
        }
    }

    total += current;  // Move to track 0
    current = 0;
    printf(" -> 0");

    for(i = 0; i < N; i++)
    {
        if(sorted[i] > head)
        {
            total += sorted[i] - current;
            current = sorted[i];

            printf(" -> %d", current);
        }
    }

    printf("\nTotal Head Movement = %d", total);
    printf("\nAverage Seek Length = %.2f\n",
           (float)total / N);

    // C-SCAN Algorithm (Direction towards 0)
    printf("\n--- C-SCAN Algorithm ---\n");

    total = 0;
    current = head;

    printf("Seek Sequence: %d", current);

    for(i = N - 1; i >= 0; i--)
    {
        if(sorted[i] < head)
        {
            total += current - sorted[i];
            current = sorted[i];

            printf(" -> %d", current);
        }
    }

    // Move to track 0
    total += current;
    current = 0;
    printf(" -> 0");

    // Circular jump to last track
    total += DISK_SIZE - 1;
    current = DISK_SIZE - 1;
    printf(" -> %d", current);

    // Continue towards lower tracks
    for(i = N - 1; i >= 0; i--)
    {
        if(sorted[i] > head)
        {
            total += current - sorted[i];
            current = sorted[i];

            printf(" -> %d", current);
        }
    }

    printf("\nTotal Head Movement = %d", total);
    printf("\nAverage Seek Length = %.2f\n",
           (float)total / N);

    return 0;
}
